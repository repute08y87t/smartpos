43:async function loadCategories() {
44:    let res = await fetch('/api/categories');
45:    let cats = await res.json();
46:    let html = '';
47:    for (let c of cats) {
48:        html += `<div class="list-item">
49:            <span>📁 ${c.name}</span>
50:            <button class="btn-yellow" style="padding:4px 12px" onclick="editCategory('${c.name}')">Edit</button>
51:            <button class="btn-red" style="padding:4px 12px" onclick="deleteCategory('${c.name}')">Delete</button>
52:        </div>`;
53:    }
54:    document.getElementById('categoryList').innerHTML = html || '<div style="color:#888;">No categories yet</div>';
55:    
56:    // Update all category dropdowns
57:    let selects = ['subCatSelect', 'filterSubCat', 'brandCatSelect', 'filterBrandCat', 'prodCatSelect'];
58:    for (let s of selects) {
59:        let sel = document.getElementById(s);
60:        if (sel) {
61:            let currentVal = sel.value;
62:            sel.innerHTML = '<option value="">Select Category</option>';
63:            for (let c of cats) {
64:                sel.innerHTML += `<option value="${c.name}">${c.name}</option>`;
65:            }
66:            if (currentVal && cats.some(c => c.name === currentVal)) sel.value = currentVal;
67:        }
68:    }
69:}

284:async function addProduct() {
285:    let cat = document.getElementById('prodCatSelect').value;
286:    let sub = document.getElementById('prodSubSelect').value;
287:    let brand = document.getElementById('prodBrandSelect').value;
288:    let size = document.getElementById('prodSize').value.trim();
289:    let price = parseFloat(document.getElementById('prodPrice').value);
290:    let cost = parseFloat(document.getElementById('prodCost').value) || 0;
291:    let stock = parseFloat(document.getElementById('prodStock').value) || 0;
292:    
293:    if (!cat || !sub || !brand) { showMessage('Select category, subcategory, and brand', 'error'); return; }
294:    if (!size) { showMessage('Enter product size', 'error'); return; }
295:    if (isNaN(price)) { showMessage('Enter selling price', 'error'); return; }
296:    
297:    let name = `${brand} ${size}`;
298:    await fetch('/api/products', {
299:        method: 'POST',
300:        headers: { 'Content-Type': 'application/json' },
301:        body: JSON.stringify({ name, size, price, cost_price: cost, stock, category: cat, subcategory: sub, brand })
302:    });
303:    
304:    document.getElementById('prodSize').value = '';
305:    document.getElementById('prodPrice').value = '';
306:    document.getElementById('prodCost').value = '';
307:    document.getElementById('prodStock').value = '';
308:    await loadProducts();
309:    showMessage('Product added successfully');
310:}
